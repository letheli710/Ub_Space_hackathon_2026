cmake_minimum_required(VERSION 3.16)
project(starship_physics LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(nlohmann_json REQUIRED)
add_library(starship-physics SHARED StarshipPhysics.cc)
target_compile_features(starship-physics PRIVATE cxx_std_17)
target_link_libraries(starship-physics PRIVATE gz-sim8::gz-sim8 gz-plugin2::register nlohmann_json::nlohmann_json)
install(TARGETS starship-physics LIBRARY DESTINATION lib)
